"""
Определитель растений + советы по уходу (Streamlit + Claude API)

Установка:
    pip install streamlit anthropic pillow

Запуск:
    export ANTHROPIC_API_KEY="ваш_ключ"      # Windows: set ANTHROPIC_API_KEY=ваш_ключ
    streamlit run plant_id.py
"""

import base64
import io
import json
import os

import anthropic
import streamlit as st
from PIL import Image

MODEL = "claude-sonnet-5-5"

PROMPT = """Ты — ботаник и эксперт по уходу за растениями. Определи растение на фото.
Верни ТОЛЬКО JSON без markdown и пояснений, на русском языке:
{"is_plant": true, "name": "русское название", "latin": "латинское название",
 "confidence": "высокая|средняя|низкая", "about": "1-2 предложения о растении",
 "care": {"light": "свет", "water": "полив и частота",
          "climate": "температура и влажность", "soil": "грунт и пересадка",
          "feed": "подкормка"},
 "problems": ["типичные проблемы и признаки"],
 "toxic": "токсичность для кошек, собак и людей",
 "alt": ["1-2 похожих варианта, если уверенность не высокая"],
 "visible_issues": "проблемы, заметные на фото (или пустая строка)"}
Если на фото нет растения: {"is_plant": false}."""


def prepare_image(file) -> str:
    """Уменьшает фото до 1024 px и возвращает base64 JPEG."""
    img = Image.open(file).convert("RGB")
    img.thumbnail((1024, 1024))
    buf = io.BytesIO()
    img.save(buf, format="JPEG", quality=85)
    return base64.standard_b64encode(buf.getvalue()).decode()


def identify(image_b64: str) -> dict:
    client = anthropic.Anthropic()  # ключ берётся из ANTHROPIC_API_KEY
    response = client.messages.create(
        model=MODEL,
        max_tokens=1500,
        messages=[{
            "role": "user",
            "content": [
                {"type": "image",
                 "source": {"type": "base64", "media_type": "image/jpeg", "data": image_b64}},
                {"type": "text", "text": PROMPT},
            ],
        }],
    )
    text = response.content[0].text.replace("```json", "").replace("```", "").strip()
    return json.loads(text[text.index("{"): text.rindex("}") + 1])


def show_result(d: dict) -> None:
    if not d.get("is_plant"):
        st.warning("Растение на фото не найдено. Снимите крупнее и при хорошем свете.")
        return

    st.header(d.get("name", "Неизвестно"))
    st.caption(f"*{d.get('latin', '')}* · Уверенность: {d.get('confidence', '—')}")
    st.write(d.get("about", ""))
    if d.get("alt"):
        st.write("Возможно также: " + ", ".join(d["alt"]))

    care = d.get("care", {})
    st.subheader("Уход")
    st.markdown(
        f"☀️ **Свет:** {care.get('light', '—')}\n\n"
        f"💧 **Полив:** {care.get('water', '—')}\n\n"
        f"🌡️ **Температура и влажность:** {care.get('climate', '—')}\n\n"
        f"🪴 **Грунт и пересадка:** {care.get('soil', '—')}\n\n"
        f"🧪 **Подкормка:** {care.get('feed', '—')}"
    )

    st.subheader("Частые проблемы")
    for p in d.get("problems", []):
        st.markdown(f"⚠️ {p}")
    if d.get("visible_issues"):
        st.info("🔍 На вашем фото: " + d["visible_issues"])
    st.markdown(f"🐾 **Токсичность:** {d.get('toxic', '—')}")


def main() -> None:
    st.set_page_config(page_title="Определитель растений", page_icon="🌿")
    st.title("🌿 Определитель растений")
    st.write("Сфотографируйте растение — получите название и советы по уходу.")

    if not os.environ.get("ANTHROPIC_API_KEY"):
        st.error("Не задана переменная окружения ANTHROPIC_API_KEY.")
        st.stop()

    source = st.radio("Источник фото", ["Загрузить файл", "Камера"], horizontal=True)
    if source == "Камера":
        photo = st.camera_input("Снимок растения")
    else:
        photo = st.file_uploader("Фото растения", type=["jpg", "jpeg", "png", "webp"])

    if photo:
        st.image(photo, use_container_width=True)
        if st.button("Определить растение", type="primary", use_container_width=True):
            with st.spinner("Анализирую…"):
                try:
                    result = identify(prepare_image(photo))
                except anthropic.APIError as e:
                    st.error(f"Ошибка API: {e}")
                    return
                except (ValueError, json.JSONDecodeError):
                    st.error("Не удалось разобрать ответ. Попробуйте ещё раз.")
                    return
            show_result(result)


if __name__ == "__main__":
    main()
